  <script>
    const CORRECT_PIN = "0807";
    let enteredPin = "";
    let score = 0;
    const bgMusic = document.getElementById('bg-music');

    function pressKey(num) {
      if (enteredPin.length < 4) {
        enteredPin += num;
      }
    }

    function checkPin() {
      if (enteredPin === CORRECT_PIN) {
        showPage('page-boxes');
        bgMusic.play();
      } else {
        enteredPin = "";
        alert("Opps! Wrong Code ❤️ Try again!");
      }
    }

    function showPage(pageId) {
      document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
      document.getElementById(pageId).classList.add('active');
    }

    function toggleMusic() {
      const recordBtn = document.getElementById('record-btn');
      if (bgMusic.paused) {
        bgMusic.play();
        recordBtn.classList.add('record-rotate');
      } else {
        bgMusic.pause();
        recordBtn.classList.remove('record-rotate');
      }
    }

    function startGame() {
      showPage('page-game');
      score = 0;
      document.getElementById('score-display').innerText = "HEARTS: 0 / 5";
      document.getElementById('win-modal').classList.remove('show');
      spawnHeart();
    }

    function spawnHeart() {
      const gameArea = document.getElementById('game-area');
      const oldHeart = document.querySelector('.catch-heart');
      if (oldHeart) oldHeart.remove();

      if (score >= 5) {
        document.getElementById('win-modal').classList.add('show');
        return;
      }

      const heart = document.createElement('div');
      heart.className = 'catch-heart';
      heart.innerHTML = '💖';
      
      const maxX = gameArea.clientWidth - 50;
      const maxY = gameArea.clientHeight - 50;
      
      const randomX = Math.max(10, Math.floor(Math.random() * maxX));
      const randomY = Math.max(40, Math.floor(Math.random() * maxY));

      heart.style.left = randomX + 'px';
      heart.style.top = randomY + 'px';

      heart.onclick = function() {
        score++;
        document.getElementById('score-display').innerText = "HEARTS: " + score + " / 5";
        spawnHeart();
      };

      gameArea.appendChild(heart);
    }
  </script>
